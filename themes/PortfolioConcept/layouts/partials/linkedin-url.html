{{- /* Turns whatever was typed in the LinkedIn field into a working address: https://www.linkedin.com/in/name used as is www.linkedin.com/in/name https:// added name or @name https://www.linkedin.com/in/name Returns "" for an empty field, so callers can hide the link. */ -}} {{- $u := trim (string .) " \t\r\n" -}} {{- $out := "" -}} {{- if $u -}} {{- $l := lower $u -}} {{- if or (hasPrefix $l "https://") (hasPrefix $l "http://") -}} {{- $out = $u -}} {{- else if in $l "linkedin.com" -}} {{- $out = printf "https://%s" (strings.TrimLeft "/" $u) -}} {{- else -}} {{- $handle := strings.TrimPrefix "@" $u -}} {{- $handle = strings.TrimPrefix "in/" (strings.Trim $handle "/") -}} {{- $out = printf "https://www.linkedin.com/in/%s" $handle -}} {{- end -}} {{- end -}} {{- return $out -}}
